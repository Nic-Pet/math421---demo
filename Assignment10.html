# Remove missing values
who <- na.omit(who)

# Make income categorical
who$income <- as.factor(trimws(who$income))

# Convert character columns to factors
who <- who %>%
  mutate(across(where(is.character), as.factor))

# Split data
set.seed(123)

index <- sample(1:nrow(who), 0.8 * nrow(who))

train <- who[index, ]
test <- who[-index, ]

# Handle categories not present in training data
for (col in names(train)) {
  if (is.factor(train[[col]]) && col != "income") {

    # Find categories that appear fewer than 10 times in training
    counts <- table(train[[col]])
    common <- names(counts[counts >= 10])

    # Group rare and unseen categories into "Other"
    train[[col]] <- as.character(train[[col]])
    test[[col]] <- as.character(test[[col]])

    train[[col]][!train[[col]] %in% common] <- "Other"
    test[[col]][!test[[col]] %in% common] <- "Other"

    # Ensure identical levels
    train[[col]] <- factor(train[[col]])
    test[[col]] <- factor(test[[col]], levels = levels(train[[col]]))
  }
}
